lemma test_shrinks:
  fixes x t :: "'a::dioid"
  assumes "t \<le> 1"
  shows "x * t \<le> x"
  by fail

lemma inequality_is_vacuous_for_zero:
  fixes P D :: "'a::dioid"
    and parallel :: "'a \<Rightarrow> 'a \<Rightarrow> 'a"
    and project :: "'a \<Rightarrow> 'b::order"
  assumes "mono project"
  shows "project (parallel P D * 0) \<le> project P"
  by fail

end
